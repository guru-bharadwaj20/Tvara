cmake_minimum_required(VERSION 3.24)
project(tvara VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(tvara_core STATIC core/src/version.cpp)
target_include_directories(tvara_core PUBLIC core/include)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_tensor tests/core/test_tensor.cpp)
  target_link_libraries(test_tensor PRIVATE tvara_core)
  add_test(NAME tensor COMMAND test_tensor)
endif()
