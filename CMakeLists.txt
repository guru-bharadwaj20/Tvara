cmake_minimum_required(VERSION 3.24)
project(tvara VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TVARA_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
add_library(tvara_options INTERFACE)
if(MSVC)
  target_compile_options(tvara_options INTERFACE /W4 /permissive-)
else()
  target_compile_options(tvara_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(TVARA_SANITIZERS)
  if(WIN32 OR MSVC)
    message(FATAL_ERROR "TVARA_SANITIZERS currently requires Linux GCC or Clang")
  endif()
  target_compile_options(tvara_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(tvara_options INTERFACE -fsanitize=address,undefined)
endif()
add_library(tvara_core STATIC core/src/version.cpp)
target_include_directories(tvara_core PUBLIC core/include)
target_link_libraries(tvara_core PUBLIC tvara_options)
include(CTest)

# Keep test assertions active in every configuration, including optimized Release.
# Enumerate after all executable targets have been declared.
if(BUILD_TESTING)
  get_property(_tvara_tests DIRECTORY PROPERTY TESTS)
  get_property(_tvara_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
  foreach(_tvara_target IN LISTS _tvara_targets)
    get_target_property(_tvara_type "${_tvara_target}" TYPE)
    get_target_property(_tvara_sources "${_tvara_target}" SOURCES)
    set(_tvara_is_test FALSE)
    if(_tvara_target IN_LIST _tvara_tests OR _tvara_target MATCHES "^(test_|prefill_t)")
      set(_tvara_is_test TRUE)
    endif()
    foreach(_tvara_source IN LISTS _tvara_sources)
      if(_tvara_source MATCHES "(^|/)tests/")
        set(_tvara_is_test TRUE)
      endif()
    endforeach()
    if(_tvara_type STREQUAL "EXECUTABLE" AND _tvara_is_test)
      if(MSVC)
        target_compile_options("${_tvara_target}" PRIVATE /UNDEBUG)
      else()
        target_compile_options("${_tvara_target}" PRIVATE -UNDEBUG)
      endif()
    endif()
  endforeach()
endif()
