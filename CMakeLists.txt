cmake_minimum_required(VERSION 3.24)
project(tvara VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TVARA_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
add_library(tvara_options INTERFACE)
if(MSVC)
  target_compile_options(tvara_options INTERFACE /W4 /permissive-)
else()
  target_compile_options(tvara_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(TVARA_SANITIZERS)
  if(WIN32 OR MSVC)
    message(FATAL_ERROR "TVARA_SANITIZERS currently requires Linux GCC or Clang")
  endif()
  target_compile_options(tvara_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(tvara_options INTERFACE -fsanitize=address,undefined)
endif()
add_library(tvara_core STATIC core/src/version.cpp)
target_include_directories(tvara_core PUBLIC core/include)
target_link_libraries(tvara_core PUBLIC tvara_options)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_tensor tests/core/test_tensor.cpp)
  target_link_libraries(test_tensor PRIVATE tvara_core)
  add_test(NAME tensor COMMAND test_tensor)
endif()

if(BUILD_TESTING)
  add_executable(test_gguf tests/core/test_gguf.cpp)
  target_link_libraries(test_gguf PRIVATE tvara_core)
  add_test(NAME gguf COMMAND test_gguf)
endif()

if(BUILD_TESTING)
  add_executable(test_quant tests/core/test_quant.cpp)
  target_link_libraries(test_quant PRIVATE tvara_core)
  add_test(NAME quant COMMAND test_quant)
endif()

if(BUILD_TESTING)
  add_executable(test_model tests/core/test_model.cpp)
  target_link_libraries(test_model PRIVATE tvara_core)
  add_test(NAME model COMMAND test_model)
endif()
